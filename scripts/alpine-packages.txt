# ===== Tier 3.5: dropbear (PC → 端末 の SSH サーバ) =====
# OpenSSH sshd は proot で privsep が破綻 (permanently_set_uid) して接続が切れる。
# dropbear は proot 下で問題なく動くため、SSH ヘルパーはこちらを起動する。
dropbear

# ===== Tier 4: mosh (UDP ローミングシェル) =====
# mosh-client は通常の UDP ソケットを使うため PRoot 配下でも動作する
# (ping/ip の raw socket/netlink とは異なり許可される)。
# mosh の launcher は perl 製なので perl が依存で入る (サイズ大)。
# musl-locales は mosh が要求する UTF-8 ロケール検出のために同梱。
mosh
musl-locales
musl-locales-lang

# ===== Tier 5: モダン CLI (初手体験向上, 約 10MB) =====
# ripgrep=高速 grep(rg), fd=高速 find, fzf=曖昧検索, bat=シンタックス付 cat,
# eza=モダン ls。vim/nano で足りるため neovim は見送り。
ripgrep
fd
fzf
bat
eza

# ===== Tier 5.5: ネットワーク診断 (小) =====
# netcat-openbsd=nc, socat=多用途リレー, mtr=traceroute+ping。
netcat-openbsd
socat
mtr
